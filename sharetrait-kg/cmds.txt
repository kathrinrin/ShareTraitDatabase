rm sharetrait-kg/sharetrait-kg-index/myindex.*

qlever-index \
  -i sharetrait-kg/sharetrait-kg-index/myindex \
  -f sharetrait-kg/sharetrait-kg.ttl \
  -F ttl


qlever-server -i sharetrait-kg/sharetrait-kg-index/myindex -p 7001


Start docker, start qleverui 

docker run -it -p 7002:7000 \
  -v "$(pwd)/db:/app/db" \
  --name qleverui \
  qleverui
